# Stub showcase. Frames A's spawn and B's patch, then drives every tool through
# synthetic mouse input (client/dev/input_script.gd). Along the way the opening guide
# ticks all four steps (claim, zone, road, power on the lot at 8,2), the road hint fires
# once (the R zoning comes before the road), and the two rejects show in player words;
# `guide` lines print that state for the log. Run with:
#   godot --path . -- --stub --input-script client/dev/scripts/stub_showcase.txt \
#     --screenshot /tmp/gamecity_stub.png --screenshot-frames 200
# Ortho sizes are metres (CameraRig.TILE = 30 m per tile).
wait 20
focus 9 8
# One wheel notch through the camera's _unhandled_input: 759 / 1.15 = ortho size 660 (22 tiles).
zoom 759
wheel 1
# Hold W for 10 frames through Input.parse_input_event, then park the pointer in the
# top edge margin for 8 frames: both pan the camera up (content slides down from under
# the crisis banner). The distance scales with frame rate (about 3 tiles at 120 Hz),
# so this is interaction evidence, not a regression value.
key W 10
edge up 8
wait 10
# Claim three neutral tiles east of A's spawn; 8,2 touches 7,2, 9,2 touches 8,2.
tool claim
click 8 2
click 9 2
click 8 3
wait 5
# Zone them: R painted by drag, C by click. 8,2 has no road yet: one "No road at 8,2" hint.
tool zone_r
drag 8 2 9 2
tool zone_c
click 8 3
wait 5
guide
# Road from inside the spawn across the new lots: two orthogonal AddEdge steps.
tool road
drag 7 2 9 2
wait 5
# Power plant on the new R lot: covers 8,2 and 9,2, so the guide completes here.
tool power
click 9 2
wait 5
guide
# Two rejects for the alert feed: B's tile (OPPONENT_IMMUTABLE), then a far tile (NOT_ADJACENT).
tool claim
click 12 2
wait 5
click 10 6
wait 5
# Demolish the C lot.
tool demolish
click 8 3
wait 5
# Tax to 20% through the slider.
slider 0.20
wait 5
# Right click clears the tool, then park the pointer on a tile so the hover highlight shows.
rclick
wait 5
tool zone_i
hover 10 3
wait 30
guide
